// 8-Bit 2-Stage Pipelined ALU in SkyWater Sky130A PDK
module alu_pipeline (
  input  wire       clk, rst,
  input  wire [7:0] a, b,
  input  wire [2:0] opcode,
  output reg  [7:0] result,
  output reg        valid
);
  reg [7:0] a_reg, b_reg, alu_result;
  reg [2:0] opcode_reg;
  reg       valid_stage1;

  // Stage 1: Input Registration on posedge clk
  always @(posedge clk or posedge rst) begin
    if (rst) begin
      a_reg <= 8'b0; b_reg <= 8'b0; opcode_reg <= 3'b0; valid_stage1 <= 1'b0;
    end else begin
      a_reg <= a; b_reg <= b; opcode_reg <= opcode; valid_stage1 <= 1'b1;
    end
  end

  // Combinational ALU Execution
  always @(*) begin
    case (opcode_reg)
      3'b000: alu_result = a_reg + b_reg; // ADD
      3'b001: alu_result = a_reg - b_reg; // SUB
      3'b010: alu_result = a_reg & b_reg; // AND
      3'b011: alu_result = a_reg | b_reg; // OR
      3'b100: alu_result = a_reg ^ b_reg; // XOR
      3'b101: alu_result = ~a_reg;        // NOT
      3'b110: alu_result = a_reg << 1;    // SLL
      3'b111: alu_result = a_reg >> 1;    // SRL
      default: alu_result = 8'b0;
    endcase
  end

  // Stage 2: Output Registration
  always @(posedge clk or posedge rst) begin
    if (rst) begin
      result <= 8'b0; valid <= 1'b0;
    end else begin
      result <= alu_result; valid <= valid_stage1;
    end
  end
endmodule

Arduino Radar System

A real-time ultrasonic object-detection radar using Arduino UNO, HC-SR04 sonar sensor, and a servo motor. Visualized in Processing IDE as a scanning PPI radar display with live distance and polar angle mapping.

Arduino UNO HC-SR04 Servo Motor Processing IDE Embedded C UART 9600
◈ HARDWARE–SOFTWARE ARCHITECTURE
  1. Servo sweeps 0°–180°, sweeping the ultrasonic transducer across the field of view in real time.
  2. HC-SR04 emits 40 kHz ultrasonic burst; echo return duration converted to distance via d = (t × 343)/2 m.
  3. Arduino serializes angle and distance telemetry at 9600 baud over UART to the host computer.
  4. Processing IDE parses the serial packet, converts polar coordinates (r, θ) to screen space, and renders target trails with phosphor persistence fade.
  5. Objects within 40 cm trigger highlighted red threat blips with live millimeter readout.

◉ PROJECT_04 // SECURITY FIRMWARE

Smart Lock Access Control

A password-secured electronic lock utilizing a 4×4 matrix keypad, LCD feedback display, EEPROM credential validation, and a servo-driven mechanical deadbolt mechanism.

Arduino UNO 4×4 Keypad LCD 16×2 Servo Deadbolt EEPROM Hashing Embedded C
◈ PASSWORD VALIDATION FLOW
  1. Matrix keypad scanned row-by-row via GPIO interrupt polling; debounced key presses buffered into input register.
  2. 16×2 display outputs immediate feedback: "ENTER PASSCODE" → masked asterisks → "ACCESS GRANTED / DENIED".
  3. Timing-safe comparison checks input against EEPROM stored master code (Default: 1234).
  4. On match: servo rotates 90° to UNLOCK state; buzzer emits dual confirmation chime; screen displays "WELCOME".
  5. Auto-relocks after 5 seconds; 3 consecutive incorrect attempts trigger a 30-second security lockdown.
